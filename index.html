Supabase Cloud Sync

Sync your folders and markdown notes across devices

Run this SQL script once in your Supabase SQL Editor to create the markdown_items table: 

create table if not exists markdown_items (
  id text primary key,
  name text not null,
  type text not null check (type in ('file', 'folder')),
  parent_id text,
  content text default '',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);
alter table markdown_items enable row level security;
create policy "public_access" on markdown_items for all using (true) with check (true);
